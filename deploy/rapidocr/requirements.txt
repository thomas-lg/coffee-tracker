# The RapidOCR reader's Python dependencies, pinned.
#
# Pinned for the same reason the NuGet lock files are, and with more at stake: the OCR
# benchmark's score is a property of these model and engine versions, so an unpinned
# rebuild could change what every scan reads without a line of this repo changing. The
# floor in OcrBenchmarkTests would then fail on a commit that touched nothing.
#
# One file, installed by both the production image and CI, so the two cannot drift from
# each other either.
#
# `rapidocr`, not `rapidocr-onnxruntime`: the latter is the same project's earlier name,
# frozen since January 2025 on PP-OCRv4. See CLAUDE.md § OCR.
#
# To bump: change a version here, rebuild, and run the benchmark. A score that moves is
# the point of doing it that way round.
rapidocr==3.9.2
onnxruntime==1.30.0

# rapidocr depends on opencv-python, whose GUI build carries ~116 MB of X11 libraries to
# draw windows a server will never open. The Dockerfile swaps it for this one afterwards;
# installing this alongside does not help, because pip honours the dependency and ships
# both.
opencv-python-headless==5.0.0.93
